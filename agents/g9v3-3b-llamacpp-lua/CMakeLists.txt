cmake_minimum_required(VERSION 3.24)
project(g9_coding_agent VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(G9_BUILD_TESTS "Build G9 unit tests" ON)
set(LUA_DIR "" CACHE PATH "Path to Lua 5.4.9 source directory (the directory containing lua.h)")
set(LLAMA_CPP_DIR "" CACHE PATH "Path to llama.cpp source directory")
set(G9_MODEL_PATH "" CACHE FILEPATH "Optional GGUF path for model smoke test")
if(NOT LUA_DIR)
  set(LUA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lua")
endif()
if(NOT EXISTS "${LUA_DIR}/lua.h" OR NOT EXISTS "${LUA_DIR}/lauxlib.h" OR NOT EXISTS "${LUA_DIR}/luaconf.h")
  message(FATAL_ERROR "Lua 5.4 source not found. Set -DLUA_DIR=/path/to/lua-5.4.9/src or populate third_party/lua.")
endif()
set(G9_LUA_SOURCES lapi.c lcode.c lctype.c ldebug.c ldo.c ldump.c lfunc.c lgc.c llex.c lmem.c lobject.c lopcodes.c lparser.c lstate.c lstring.c ltable.c ltm.c lundump.c lvm.c lzio.c lauxlib.c lbaselib.c lmathlib.c lstrlib.c ltablib.c lutf8lib.c)
set(_lua_sources)
foreach(_src IN LISTS G9_LUA_SOURCES)
  if(NOT EXISTS "${LUA_DIR}/${_src}")
    message(FATAL_ERROR "Lua source file missing: ${LUA_DIR}/${_src}")
  endif()
  list(APPEND _lua_sources "${LUA_DIR}/${_src}")
endforeach()
add_library(g9_lua_vendor STATIC ${_lua_sources})
target_include_directories(g9_lua_vendor PUBLIC "${LUA_DIR}")
if(UNIX)
  target_link_libraries(g9_lua_vendor PUBLIC m)
endif()
if(NOT LLAMA_CPP_DIR)
  set(LLAMA_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp")
endif()
if(NOT EXISTS "${LLAMA_CPP_DIR}/CMakeLists.txt" OR NOT EXISTS "${LLAMA_CPP_DIR}/include/llama.h")
  message(FATAL_ERROR "llama.cpp source not found. Set -DLLAMA_CPP_DIR=/path/to/llama.cpp at commit f3f1a8f2760f28325a5ec20c05b171e5b7c83a29 or populate third_party/llama.cpp.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_DIR}" "${CMAKE_BINARY_DIR}/llama.cpp" EXCLUDE_FROM_ALL)
add_library(g9_core STATIC src/json.cpp src/contract.cpp)
target_include_directories(g9_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(g9_lua STATIC src/lua_agent.cpp)
target_link_libraries(g9_lua PUBLIC g9_core g9_lua_vendor)
target_include_directories(g9_lua PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(g9_tools STATIC src/workspace.cpp src/guardrails.cpp src/patch_engine.cpp src/tools.cpp src/tool_registry.cpp src/process.cpp src/process_sandbox.cpp src/command_tool.cpp)
target_link_libraries(g9_tools PUBLIC g9_core)
target_include_directories(g9_tools PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(g9_protocol STATIC src/tool_protocol.cpp)
target_link_libraries(g9_protocol PUBLIC g9_core g9_tools)
target_include_directories(g9_protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(g9_runtime STATIC src/llama_runtime.cpp)
target_link_libraries(g9_runtime PUBLIC g9_core g9_lua llama)
target_include_directories(g9_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(g9_runtime PUBLIC G9_LLAMA_CPP_REVISION="f3f1a8f2760f28325a5ec20c05b171e5b7c83a29")
add_library(g9_loop STATIC src/agent_loop.cpp)
target_link_libraries(g9_loop PUBLIC g9_core g9_tools g9_protocol)
target_include_directories(g9_loop PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(g9_ui STATIC src/cli.cpp src/terminal.cpp src/line_editor.cpp)
target_link_libraries(g9_ui PUBLIC g9_core g9_tools g9_loop)
target_include_directories(g9_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_library(g9_agentlib STATIC src/application.cpp)
target_link_libraries(g9_agentlib PUBLIC g9_core g9_lua g9_tools g9_protocol g9_runtime g9_loop g9_ui)
target_include_directories(g9_agentlib PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(g9_agentlib PUBLIC G9_LLAMA_CPP_REVISION="f3f1a8f2760f28325a5ec20c05b171e5b7c83a29")
add_executable(g9-agent src/main.cpp)
target_link_libraries(g9-agent PRIVATE g9_agentlib)
foreach(_target g9_core g9_lua g9_tools g9_protocol g9_runtime g9_loop g9_ui g9_agentlib g9-agent)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(G9_BUILD_TESTS)
  enable_testing()
  function(g9_add_test name source)
    add_executable(${name} "tests/${source}")
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests" "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(${name} PRIVATE ${ARGN})
    add_test(NAME ${name} COMMAND ${name})
  endfunction()
  g9_add_test(g9_test_json test_json.cpp g9_core)
  g9_add_test(g9_test_contract test_contract.cpp g9_core)
  g9_add_test(g9_test_workspace test_workspace.cpp g9_tools)
  g9_add_test(g9_test_guardrails test_guardrails.cpp g9_tools)
  g9_add_test(g9_test_patch_engine test_patch_engine.cpp g9_tools)
  g9_add_test(g9_test_tools test_tools.cpp g9_tools)
  g9_add_test(g9_test_process test_process.cpp g9_tools)
  g9_add_test(g9_test_command_tool test_command_tool.cpp g9_tools)
  g9_add_test(g9_test_tool_protocol test_tool_protocol.cpp g9_protocol)
  g9_add_test(g9_test_agent_loop test_agent_loop.cpp g9_loop)
  g9_add_test(g9_test_cli test_cli.cpp g9_ui)
  g9_add_test(g9_test_lua_agent test_lua_agent.cpp g9_lua)
  if(G9_MODEL_PATH AND EXISTS "${G9_MODEL_PATH}")
    g9_add_test(g9_model_smoke test_model_smoke.cpp g9_runtime)
    target_compile_definitions(g9_model_smoke PRIVATE G9_MODEL_PATH="${G9_MODEL_PATH}")
    set_tests_properties(g9_model_smoke PROPERTIES LABELS model)
  endif()
endif()
